//@version=5
indicator("XAUUSD Zones (Analyzer v7.2)", overlay=true, max_lines_count=500, max_labels_count=500)

strongR_input = input.string("", "Strongest Resistances (comma-separated)", group="Strongest")
strongS_input = input.string("", "Strongest Supports (comma-separated)", group="Strongest")
nearR_input   = input.string("", "Nearest Resistances (comma-separated)", group="Nearest")
nearS_input   = input.string("", "Nearest Supports (comma-separated)", group="Nearest")

showLabels = input.bool(true, "Show Labels", group="Display")
colStrongR = input.color(color.new(color.red, 0), "Strong R color", group="Colors", inline="s")
colStrongS = input.color(color.new(color.green, 0), "Strong S color", group="Colors", inline="s")
colNearR   = input.color(color.new(color.orange, 0), "Near R color", group="Colors", inline="n")
colNearS   = input.color(color.new(color.aqua, 0), "Near S color", group="Colors", inline="n")

parsePrices(s) =>
    result = array.new_float()
    if str.length(s) > 0
        parts = str.split(s, ",")
        for i = 0 to array.size(parts) - 1
            cleaned = str.replace_all(str.trim(array.get(parts, i)), " ", "")
            if str.length(cleaned) > 0
                val = str.tonumber(cleaned)
                if not na(val)
                    array.push(result, val)
    result

strongR = parsePrices(strongR_input)
strongS = parsePrices(strongS_input)
nearR   = parsePrices(nearR_input)
nearS   = parsePrices(nearS_input)

drawLines(arr, col, width, style, prefix) =>
    if array.size(arr) > 0
        for i = 0 to array.size(arr) - 1
            p = array.get(arr, i)
            line.new(bar_index, p, bar_index + 500, p, color=col, width=width, style=style, extend=extend.both)
            if showLabels
                label.new(bar_index + 500, p, prefix + ": " + str.tostring(p, "#.##"), color=color.new(col, 70), textcolor=col, style=label.style_label_left, size=size.small)

if barstate.islast
    drawLines(strongR, colStrongR, 3, line.style_solid, "R-strong")
    drawLines(strongS, colStrongS, 3, line.style_solid, "S-strong")
    drawLines(nearR, colNearR, 1, line.style_dashed, "R-near")
    drawLines(nearS, colNearS, 1, line.style_dashed, "S-near")

atr14 = ta.atr(14)
checkProximity(arr) =>
    result = false
    if array.size(arr) > 0
        for i = 0 to array.size(arr) - 1
            p = array.get(arr, i)
            if math.abs(close - p) <= 0.3 * atr14
                result := true
                break
    result

alertcondition(checkProximity(strongR), "Near Strong Resistance", "Price near strong resistance")
alertcondition(checkProximity(strongS), "Near Strong Support", "Price near strong support")
alertcondition(checkProximity(nearR), "Near Resistance", "Price near resistance")
alertcondition(checkProximity(nearS), "Near Support", "Price near support")
